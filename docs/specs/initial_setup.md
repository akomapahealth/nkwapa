	•	packages/db/prisma/seed.ts (seed + clinic + research settings)

2) packages/db/prisma/seed.ts

This seed is intentionally minimal: creates a clinic, a system admin user (if you provide SEED_SYSTEM_ADMIN_SUB), and default research settings. It won’t break if env vars are missing.

// packages/db/prisma/seed.ts
import { PrismaClient, UserRole } from "@prisma/client";

const prisma = new PrismaClient();

async function main() {
  const clinicName = process.env.SEED_CLINIC_NAME ?? "Nkwapa Clinic - Demo";
  const clinicRegion = process.env.SEED_CLINIC_REGION ?? "Greater Accra";
  const clinicCountry = process.env.SEED_CLINIC_COUNTRY ?? "GH";

  const clinic = await prisma.clinic.upsert({
    where: { name: clinicName },
    update: { region: clinicRegion, countryCode: clinicCountry, isActive: true },
    create: { name: clinicName, region: clinicRegion, countryCode: clinicCountry, isActive: true },
  });

  const sysAdminSub = process.env.SEED_SYSTEM_ADMIN_SUB;
  const sysAdminName = process.env.SEED_SYSTEM_ADMIN_NAME ?? "System Admin";

  if (sysAdminSub) {
    const user = await prisma.user.upsert({
      where: { keycloakSub: sysAdminSub },
      update: { displayName: sysAdminName, isActive: true },
      create: { keycloakSub: sysAdminSub, displayName: sysAdminName, isActive: true },
    });

    // Global SYSTEM_ADMIN role (clinicId = null)
    await prisma.userClinicRole.upsert({
      where: { userId_clinicId_role: { userId: user.id, clinicId: null, role: UserRole.SYSTEM_ADMIN } },
      update: {},
      create: { userId: user.id, clinicId: null, role: UserRole.SYSTEM_ADMIN },
    });

    // Also give a DIRECTOR role for demo on the demo clinic (optional convenience)
    await prisma.userClinicRole.upsert({
      where: { userId_clinicId_role: { userId: user.id, clinicId: clinic.id, role: UserRole.DIRECTOR } },
      update: {},
      create: { userId: user.id, clinicId: clinic.id, role: UserRole.DIRECTOR },
    });

    // Default research settings for clinic
    await prisma.clinicResearchSettings.upsert({
      where: { clinicId: clinic.id },
      update: { updatedByUserId: user.id, researchEnabled: false, requiresDirectorApprovalEachExport: true },
      create: {
        clinicId: clinic.id,
        updatedByUserId: user.id,
        researchEnabled: false,
        requiresDirectorApprovalEachExport: true,
      },
    });

    console.log("Seeded clinic + system admin user + roles + clinic research settings.");
  } else {
    // Create disabled research settings without updatedBy (needs a user), so skip.
    console.log("SEED_SYSTEM_ADMIN_SUB not provided; seeded clinic only. Research settings will be created after first Director exists.");
  }

  console.log({ clinicId: clinic.id, clinicName: clinic.name });
}

main()
  .catch((e) => {
    console.error(e);
    process.exit(1);
  })
  .finally(async () => {
    await prisma.$disconnect();
  });

7) GitHub Actions CI: .github/workflows/ci.yml

This workflow runs install/lint/typecheck/test/build and validates Prisma migrations against a real Postgres service.

# .github/workflows/ci.yml
name: CI

on:
  pull_request:
  push:
    branches: [ main ]

jobs:
  build-test:
    runs-on: ubuntu-latest
    services:
      postgres:
        image: postgres:16
        env:
          POSTGRES_DB: nkwapa
          POSTGRES_USER: nkwapa
          POSTGRES_PASSWORD: nkwapa
        ports:
          - 5432:5432
        options: >-
          --health-cmd="pg_isready -U nkwapa -d nkwapa"
          --health-interval=5s
          --health-timeout=3s
          --health-retries=20

    env:
      DATABASE_URL: postgresql://nkwapa:nkwapa@localhost:5432/nkwapa?schema=public

    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Setup Node
        uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: "npm"

      - name: Install
        run: npm ci

      - name: Lint
        run: npm run lint --workspaces

      - name: Typecheck
        run: npm run typecheck --workspaces

      - name: Test
        run: npm run test --workspaces

      - name: Build
        run: npm run build --workspaces

      - name: Prisma generate
        run: npm run db:generate

      - name: Prisma migrate deploy (validate)
        run: npm run db:migrate:deploy

If you’re using pnpm or turborepo, agents should adjust the install/cache commands accordingly. The above is correct for plain npm workspaces.

10) Phone input UI (country selector locked to Ghana)

Since you explicitly want the selector locked to Ghana, the frontend spec should enforce:
	•	country = GH always (no dropdown interaction)
	•	input accepts local Ghana forms, but UI shows it as Ghana
	•	backend normalizes to E.164 regardless

When your UI agent builds it, tell them to use a phone input component that supports:
	•	default country = GH
	•	disable country dropdown

Examples: react-phone-number-input or mui-tel-input (both can be configured accordingly).
